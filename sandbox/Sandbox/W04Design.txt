Foundation #2: Encapsulation with Online Ordering

class: Product
    Attributes:
        _name: string
        _productid: int
        _price: float
        _quantity: int
    Methods:
        GetTotalCost(): float

----------------------------------------------------------------

class: Address
    Attributes:
        _street: string
        _city: string
        _stateProvince: string
        _country: string
    Methods:
        IsInUSA(): bool
        GetFullAddress(): string

----------------------------------------------------------------

class: Customer
    Attributes:
        _name: string
        _address: Address
    Methods:
        IsInUSA(): bool

----------------------------------------------------------------

class: Order
    Attributes:
        _product: List<Product>
        _customer: Customer
    Methods:
        GetTotalCost(): float
        GetPackagingLabel(): string
        GetShippingLabel(): string
